cmake_minimum_required(VERSION 3.12)
project(one_wire)

# Add OneWire source files
add_library(one_wire STATIC src/OneWire.c)

# Include OneWire header files
# IMPORTANTE: El nombre del target debe coincidir: "one_wire" (en minúsculas)
target_include_directories(one_wire PUBLIC
    ${CMAKE_CURRENT_LIST_DIR}/include
)

# Add dependencies
target_link_libraries(one_wire PUBLIC pico_stdlib hardware_gpio)
